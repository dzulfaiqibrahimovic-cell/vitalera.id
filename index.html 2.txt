        function calculateQuiz() {
            if (Object.keys(quizAnswers).length < quizQuestions.length) {

            const percent = Math.round((total / (quizQuestions.length * 5)) * 100);

            document.getElementById('check-score').textContent = percent;
            document.getElementById('check-progress').style.width = percent + '%';

        let chatHistory = [

        async function sendChat() {
            const input = document.getElementById('chat-input');
            const message = input.value.trim();
            if (!message) return;

            const chatBox = document.getElementById('chat-box');
            const sendBtn = document.getElementById('chat-send-btn');

            const userMessage = document.createElement('div');
            userMessage.className = 'chat-message flex justify-end';
            userMessage.innerHTML = `<div class="bg-vital-700 text-white rounded-2xl rounded-tr-sm px-4 py-3 text-sm leading-6 max-w-[82%]">${escapeHTML(message)}</div>`;
            chatBox.appendChild(userMessage);

            chatHistory.push({ role: "user", content: message });

            input.value = '';
            input.disabled = true;
            sendBtn.disabled = true;
            chatBox.scrollTop = chatBox.scrollHeight;

            const typingEl = document.createElement('div');
            typingEl.className = 'chat-message flex gap-2';
            typingEl.id = 'typing-indicator';
            typingEl.innerHTML = `
                <div class="w-8 h-8 rounded-full bg-vital-100 text-vital-700 flex items-center justify-center flex-shrink-0"><i data-lucide="bot" class="w-4 h-4"></i></div>
                <div class="bg-gray-50 rounded-2xl rounded-tl-sm px-4 py-3 text-sm leading-6"><span class="typing-dot"></span><span class="typing-dot"></span><span class="typing-dot"></span></div>
            `;
            chatBox.appendChild(typingEl);
            lucide.createIcons();
            chatBox.scrollTop = chatBox.scrollHeight;

            try {
                const response = await fetch("https://api.anthropic.com/v1/messages", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({
                        model: "claude-sonnet-4-6",
                        max_tokens: 1000,
                        system: VITA_BUDDY_SYSTEM_PROMPT,
                        messages: chatHistory
                    })
                });

                const data = await response.json();
                const textBlock = (data.content || []).find(b => b.type === "text");
                const reply = textBlock ? textBlock.text : "Maaf, aku belum bisa menjawab itu sekarang. Coba tanyakan dengan cara lain, ya.";

                chatHistory.push({ role: "assistant", content: reply });

                document.getElementById('typing-indicator')?.remove();

                const aiMessage = document.createElement('div');
                aiMessage.className = 'chat-message flex gap-2';
                aiMessage.innerHTML = `
                    <div class="w-8 h-8 rounded-full bg-vital-100 text-vital-700 flex items-center justify-center flex-shrink-0"><i data-lucide="bot" class="w-4 h-4"></i></div>
                    <div class="bg-gray-50 rounded-2xl rounded-tl-sm px-4 py-3 text-sm leading-6 max-w-[82%]">${formatReply(reply)}</div>
                `;
                chatBox.appendChild(aiMessage);
                lucide.createIcons();
                chatBox.scrollTop = chatBox.scrollHeight;

            } catch (err) {
                document.getElementById('typing-indicator')?.remove();
                const errMessage = document.createElement('div');
                errMessage.className = 'chat-message flex gap-2';
                errMessage.innerHTML = `
                    <div class="w-8 h-8 rounded-full bg-vital-100 text-vital-700 flex items-center justify-center flex-shrink-0"><i data-lucide="bot" class="w-4 h-4"></i></div>
                    <div class="bg-gray-50 rounded-2xl rounded-tl-sm px-4 py-3 text-sm leading-6 max-w-[82%]">Maaf, koneksi ke Vita Buddy sedang terganggu. Coba lagi sebentar lagi, ya 🙏</div>
                `;
                chatBox.appendChild(errMessage);
                lucide.createIcons();
            } finally {
                input.disabled = false;
                sendBtn.disabled = false;
                input.focus();
                chatBox.scrollTop = chatBox.scrollHeight;
            }
        }

        function formatReply(text) {
            const escaped = escapeHTML(text);
            return escaped.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>').replace(/\n/g, '<br>');
        }

        function escapeHTML(text) {
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }
